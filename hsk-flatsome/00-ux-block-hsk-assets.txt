(function () {
function norm(s) { s = (s || '').toLowerCase().normalize('NFD'); var o = ''; for (var i = 0; i !== s.length; i++) { var c = s.charCodeAt(i); if (c >= 768 && 879 >= c) continue; o += s[i]; } return o.replace(/đ/g, 'd').replace(/[ ]+/g, ' ').trim(); }
function el(tag, cls, text) { var n = document.createElement(tag); if (cls) n.className = cls; if (text != null) n.textContent = text; return n; }
function each(list, fn) { Array.prototype.forEach.call(list, fn); }
function pad(n) { return n > 9 ? String(n) : '0' + n; }

function readCards() {
var out = [], seen = {};
each(document.querySelectorAll('.hs-src .ts-portal-card'), function (card) {
var h = card.querySelector('.ts-portal-card-title'), a = card.querySelector('a[href]');
if (!h || !a || seen[a.href]) return;
seen[a.href] = 1;
var t = norm(h.textContent), d = { href: a.href, skill: 'full', level: '', no: 0 };
MATCH.forEach(function (k) { if (d.skill === 'full' && SK[k].re.test(t)) d.skill = k; });
var lv = t.match(/hsk ?([1-6])($|[^0-9])/);
if (lv) d.level = 'hsk' + lv[1];
var m = t.match(/(?:de|test)[ ]*(?:so[ ]*)?#?[ ]*([0-9]+)/);
d.no = m ? parseInt(m[1], 10) : 0;
out.push(d);
});
return out;
}

}
function buildExams(box, cards, limit) {
var level = box.getAttribute('data-level'), name = 'HSK ' + level.replace('hsk', '');
var mine = cards.filter(function (d) { return d.level === level; });
var nums = [];
mine.forEach(function (d) { if (nums.indexOf(d.no) === -1) nums.push(d.no); });
nums.sort(function (a, b) { return a - b; });
if (limit) nums = nums.slice(0, limit);
while (box.firstChild) box.removeChild(box.firstChild);
if (!nums.length) { box.appendChild(el('div', 'hs-empty', 'Đề ' + name + ' đang được cập nhật. Bạn quay lại sau nhé!')); return 0; }
nums.forEach(function (n) {
var set = mine.filter(function (d) { return d.no === n; });
function pick(sk) { for (var i = 0; i !== set.length; i++) if (set[i].skill === sk) return set[i]; return null; }
var art = el('article', 'hs-exam');
art.style.setProperty('--c', COLORS[level.replace('hsk', '')] || '#0d3f38');
var head = el('header', 'hs-exam-h');
head.appendChild(el('span', 'hs-num', n ? pad(n) : '★'));
var hd = el('div');
hd.appendChild(el('h3', '', (n ? 'Đề ' + n + ' thi thử ' : 'Đề thi thử ') + name));
hd.appendChild(el('p', '', box.getAttribute('data-desc') || ''));
head.appendChild(hd);
art.appendChild(head);
var rows = el('div', 'hs-rows'), cnt = 0;
ORDER.forEach(function (sk) { var d = pick(sk); if (d) { rows.appendChild(row(sk, d)); cnt++; } });
if (cnt) art.appendChild(rows);
var f = pick('full');
if (f) {
var foot = el('div', 'hs-foot');
var b = el('a', 'hs-full'); b.href = f.href;
b.appendChild(el('i', 'fa-solid fa-circle-play'));
b.appendChild(document.createTextNode(' Thi Full đề'));
foot.appendChild(b);
art.appendChild(foot);
}
box.appendChild(art);
});
return nums.length;
}
function openSkill(panel, cards, sk) {

}
