/* VSTEP – dựng danh sách đề từ dữ liệu TestSpace ([tqp_category_landing category="..."]) đặt trong cột .vs-src.
   Tên đề nên chứa kỹ năng (Listening/Reading/Writing/Speaking) và bậc (B1/B2/C1), ví dụ: "VSTEP Văn Lang B2 - Listening - Đề 01".
   Đề không có tên kỹ năng được coi là đề FULL 4 kỹ năng.
   Không dùng ký tự "nhỏ hơn" và dòng trống để WordPress không làm hỏng mã. */
(function () {
  function norm(s) { s = (s || '').toLowerCase().normalize('NFD'); var o = ''; for (var i = 0; i !== s.length; i++) { var c = s.charCodeAt(i); if (c >= 768 && 879 >= c) continue; o += s[i]; } return o.replace(/đ/g, 'd').replace(/\s+/g, ' ').trim(); }
  function el(tag, cls, text) { var n = document.createElement(tag); if (cls) n.className = cls; if (text != null) n.textContent = text; return n; }
  function each(list, fn) { Array.prototype.forEach.call(list, fn); }
  function fmt(n) { return Number(n || 0).toLocaleString('vi-VN'); }
  function pad(n) { return n > 9 ? String(n) : '0' + n; }
  function setStat(name, val) { each(document.querySelectorAll('[data-vs-stat="' + name + '"]'), function (n) { n.textContent = val; }); }
  var SKILL = {
    listening: { name: 'Listening', ico: '🎧', re: /listening|nghe hieu|ky nang nghe/ },
    reading: { name: 'Reading', ico: '📖', re: /reading|doc hieu|ky nang doc/ },
    writing: { name: 'Writing', ico: '✍️', re: /writing|ky nang viet|bai viet/ },
    speaking: { name: 'Speaking', ico: '🎙️', re: /speaking|ky nang noi/ }
  };
  var ORDER = ['listening', 'reading', 'writing', 'speaking', 'full'];
  var LEVELS = ['b1', 'b2', 'c1'];
  function readCards(scope) {
    var out = [], seen = {};
    each((scope || document).querySelectorAll('.vs-src .ts-portal-card'), function (card) {
      var h = card.querySelector('.ts-portal-card-title'), a = card.querySelector('a[href]');
      if (!h || !a || seen[a.href]) return;
      seen[a.href] = 1;
      var d = { title: h.textContent.replace(/\s+/g, ' ').trim(), href: a.href, min: 0, q: 0 };
      each(card.querySelectorAll('.ts-portal-meta-item'), function (it) {
        var s = it.querySelector('strong'), n = s ? parseInt(s.textContent, 10) || 0 : 0, t = norm(it.textContent);
        if (t.indexOf('phut') !== -1) d.min = n; else if (t.indexOf('cau') !== -1) d.q = n;
      });
      var t = norm(d.title);
      d.skill = 'full';
      Object.keys(SKILL).forEach(function (k) { if (d.skill === 'full' && SKILL[k].re.test(t)) d.skill = k; });
      var lv = t.match(/(^|[^a-z0-9])(b1|b2|c1)([^a-z0-9]|$)/);
      d.level = lv ? lv[2] : '';
      var m = t.match(/(?:de|test)\s*(?:so\s*)?#?\s*(\d+)/) || t.match(/(\d+)\s*$/);
      d.no = m ? parseInt(m[1], 10) : 0;
      out.push(d);
    });
    out.sort(function (a, b) { return (ORDER.indexOf(a.skill) - ORDER.indexOf(b.skill)) || (LEVELS.indexOf(a.level) - LEVELS.indexOf(b.level)) || (a.no - b.no) || a.title.localeCompare(b.title, 'vi'); });
    return out;
  }
  function group(all) { var by = { listening: [], reading: [], writing: [], speaking: [], full: [] }; all.forEach(function (d) { by[d.skill].push(d); }); return by; }
  function label(d) {
    var lv = d.level ? d.level.toUpperCase() + ' ' : '';
    if (d.no) return (d.skill === 'full' ? 'Đề thi thử ' : SKILL[d.skill].name + ' ') + lv + '– Đề ' + pad(d.no);
    return d.title;
  }
  function item(d) {
    var a = el('a', 'vs-item');
    a.href = d.href;
    a.title = d.title;
    a.setAttribute('data-level', d.level || 'none');
    var body = el('span', 'vs-item-body');
    body.appendChild(el('span', 'vs-item-k', 'VSTEP' + (d.skill !== 'full' ? ' · ' + SKILL[d.skill].name : '') + (d.level ? ' · ' + d.level.toUpperCase() : '')));
    body.appendChild(el('span', 'vs-item-t', label(d)));
    var meta = [];
    if (d.q) meta.push(d.q + ' câu');
    if (d.min) meta.push(d.min + ' phút');
    if (meta.length) body.appendChild(el('span', 'vs-item-m', meta.join(' · ')));
    a.appendChild(body);
    if (d.min) a.appendChild(el('span', 'vs-item-min', d.min + "'"));
    a.appendChild(el('span', 'vs-chev', '›'));
    return a;
  }
  function clear(n) { while (n.firstChild) n.removeChild(n.firstChild); }
  /* Danh sách đề của 1 kỹ năng + bộ lọc bậc B1/B2/C1 (chỉ hiện khi có từ 2 bậc trở lên) */
  function fillList(box, list) {
    clear(box);
    var panel = box.closest('.vs-panel');

  function initSkillTabs(by) {
    each(document.querySelectorAll('.vs-skilltabs'), function (wrap) {
      var tabs = wrap.querySelectorAll('.vs-skilltab'), scope = wrap.parentNode;
      function open(sk, scroll) {
        each(tabs, function (t) { t.classList.toggle('is-active', t.getAttribute('data-skill') === sk); });
        each(scope.querySelectorAll('.vs-panel[data-skill]'), function (p) { if (p.getAttribute('data-skill') === sk) p.removeAttribute('hidden'); else p.setAttribute('hidden', ''); });
        if (scroll) { var p = scope.querySelector('.vs-panel[data-skill="' + sk + '"]'); if (p && p.getBoundingClientRect().top > window.innerHeight * 0.8) window.scrollTo({ top: p.getBoundingClientRect().top + window.pageYOffset - 120, behavior: 'smooth' }); }
      }
      each(tabs, function (t) {
        var sk = t.getAttribute('data-skill'), n = (by[sk] || []).length;
        if (!t.querySelector('.vs-st-n')) t.appendChild(el('span', 'vs-st-n', '(' + n + ')'));
        if (sk === 'full' && !n) t.style.display = 'none';
        if (!n) t.classList.add('is-empty');
        t.setAttribute('role', 'button');
        t.setAttribute('tabindex', '0');
        t.addEventListener('click', function () { open(sk, true); });
        t.addEventListener('keydown', function (e) { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); open(sk, true); } });
      });
      var first = null;
      each(tabs, function (t) { if (!first && (by[t.getAttribute('data-skill')] || []).length) first = t.getAttribute('data-skill'); });
      var hash = (location.hash || '').replace('#', '');
      open(by[hash] && by[hash].length ? hash : (first || 'listening'), false);
    });
  }
